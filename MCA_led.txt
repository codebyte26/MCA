#include<reg51.h>
unsigned char led_pattern=0x01;
unsigned int timer_count=0;
void timer0_ISR(void)interrupt 1
{
	TH0=0XD8;
	TL0=0XF0;
	timer_count++;
	if(timer_count>=10)
	{
		timer_count=0;
		P1=led_pattern;
		led_pattern=led_pattern<<1;
		if(led_pattern==0x00)
		{
			led_pattern=0x01;
		}
	}
}

	{
	}
}
