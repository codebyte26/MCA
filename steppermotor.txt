#include <reg51.h>

#define MOTOR P2

void delay(unsigned int ms)
{
    unsigned int i, j;

    for(i = 0; i < ms; i++)
    {
        for(j = 0; j < 120; j++);
    }
}

void clockwise(void)
{
    MOTOR = 0x09;
    delay(10);

    MOTOR = 0x0C;
    delay(10);

    MOTOR = 0x06;
    delay(10);

    MOTOR = 0x03;
    delay(10);
}

void anticlockwise(void)
{
    MOTOR = 0x03;
    delay(10);

    MOTOR = 0x06;
    delay(10);

    MOTOR = 0x0C;
    delay(10);

    MOTOR = 0x09;
    delay(10);
}

void main(void)
{
    unsigned int i;

    MOTOR = 0x00;

    while(1)
    {
        // Clockwise rotation
        for(i = 0; i < 100; i++)
        {
            clockwise();
        }

        delay(500);

        // Anticlockwise rotation
        for(i = 0; i < 100; i++)
        {
            anticlockwise();
        }

        delay(500);
    }
}
